        // Add dynamic background color for hero-p based on background
        function getContrastingColor(rgbColor) {
            const values = rgbColor.match(/\d+/g);
            if (!values || values.length < 3) return 'rgba(0, 0, 0, 0.4)';

            const [r, g, b] = values.map(Number);

            // Simple luminance calculation to determine if the color is light or dark
            const luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255;

            // Return a semi-transparent black for light backgrounds, or a semi-transparent white for dark ones
            return luminance > 0.5 ? 'rgba(0, 0, 0, 0.3)' : 'rgba(255, 255, 255, 0.3)';
        }
